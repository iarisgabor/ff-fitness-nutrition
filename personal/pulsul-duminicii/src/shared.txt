function esc(s){
  return String(s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}

const tt = document.getElementById('tooltip');
function showTip(html, x, y){
  if(!tt) return;
  tt.innerHTML = html;
  tt.classList.add('show');
  positionTip(x,y);
}
function positionTip(x,y){
  if(!tt) return;
  const pad = 16;
  let left = x + 16, top = y + 16;
  const rect = tt.getBoundingClientRect();
  if(left + rect.width + pad > window.innerWidth) left = x - rect.width - 16;
  if(top + rect.height + pad > window.innerHeight) top = y - rect.height - 16;
  tt.style.left = left + 'px';
  tt.style.top = top + 'px';
}
let tipOwner = null;
function hideTip(){ tipOwner = null; if(tt) tt.classList.remove('show'); }
// Mouse: apare la hover, dispare la ieșire. Deget: o atingere îl arată, a doua atingere
// pe același element, o atingere în altă parte sau derularea îl ascund (înainte rămânea
// pe ecran până atingeai alt element cu tooltip).
function bindTip(el, htmlFn){
  el.addEventListener('pointermove', e => { if (e.pointerType === 'mouse') showTip(htmlFn(), e.clientX, e.clientY); });
  el.addEventListener('pointerleave', e => { if (e.pointerType === 'mouse') hideTip(); });
  el.addEventListener('pointerup', e => {
    if (e.pointerType === 'mouse') return;
    if (tipOwner === el) { hideTip(); return; }
    showTip(htmlFn(), e.clientX, e.clientY);
    tipOwner = el;
  });
}
document.addEventListener('pointerdown', e => {
  if (e.pointerType !== 'mouse' && tipOwner && !tipOwner.contains(e.target)) hideTip();
}, true);
window.addEventListener('scroll', () => { if (tipOwner) hideTip(); }, {passive:true});

function isDarkNow(){
  const forced = document.documentElement.getAttribute('data-theme');
  if (forced) return forced === 'dark';
  return window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
}
function setTheme(theme){ // 'light' | 'dark'
  document.documentElement.setAttribute('data-theme', theme);
  // bara de stare a telefonului urmează tema aleasă, nu doar pe cea a sistemului
  document.querySelectorAll('meta[name=theme-color]').forEach(m => { m.content = theme === 'dark' ? '#090C13' : '#F1F3F8'; });
  try { localStorage.setItem('puls-theme', theme); } catch(e){}
}
function initThemeToggle(){
  const btn = document.getElementById('theme-btn');
  if (!btn) return;
  btn.addEventListener('click', () => setTheme(isDarkNow() ? 'light' : 'dark'));
}

// "Beni Oz" -> "BO", "BisericaLogos" -> "BL", "Marius" -> "M"
function initials(name){
  const words = String(name || '').trim().split(/\s+/).filter(Boolean);
  if (words.length > 1) return (words[0][0] + words[1][0]).toUpperCase();
  const w = words[0] || '?';
  const caps = w.slice(1).match(/[A-ZĂÂÎȘȚ]/);
  return (w[0] + (caps ? caps[0] : '')).toUpperCase();
}

// ---------------------------------------------------------------- panouri (sheet)

// <dialog> nativ: focusul rămâne înăuntru, Esc închide, restul paginii devine inert.
// Pe telefon urcă de jos, cu mâner — se închide și trăgând în jos; pe desktop e centrat.
function openSheet(html, opts){
  opts = opts || {};
  const dlg = document.createElement('dialog');
  dlg.className = 'sheet' + (opts.className ? ' ' + opts.className : '');
  if (opts.label || opts.title) dlg.setAttribute('aria-label', opts.label || opts.title);
  dlg.innerHTML = `<div class="sheet-panel">
      <div class="sheet-grab"><div class="sheet-handle" aria-hidden="true"></div>${opts.title ? `<div class="sheet-title">${esc(opts.title)}</div>` : ''}</div>
      <div class="sheet-content">${html}</div>
    </div>`;
  dlg._onClose = opts.onClose || null;
  document.body.appendChild(dlg);
  document.documentElement.classList.add('sheet-open');
  dlg.showModal();
  dlg.addEventListener('click', (e) => { if (e.target === dlg) closeSheet(dlg); }); // clic pe fundal
  dlg.addEventListener('cancel', (e) => { e.preventDefault(); closeSheet(dlg); }); // Esc
  enableSheetSwipe(dlg);
  return dlg;
}
function closeSheet(dlg, value){
  if (!dlg || !dlg.open || dlg._closing) return;
  dlg._closing = true;
  dlg.classList.add('closing');
  const done = () => {
    dlg.close();
    dlg.remove();
    // derularea rămâne blocată cât mai e deschis alt panou (ex. confirmare peste panoul de editare)
    if (!document.querySelector('dialog.sheet[open], .drawer:not([hidden])')) document.documentElement.classList.remove('sheet-open');
    if (dlg._onClose) dlg._onClose(value);
  };
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) done();
  else setTimeout(done, 170);
}
function enableSheetSwipe(dlg){
  enableSwipeDown(dlg.querySelector('.sheet-grab'), dlg.querySelector('.sheet-panel'), () => closeSheet(dlg));
}
// Tragere în jos de mâner (doar deget): panoul urmează degetul și se închide peste 90px.
function enableSwipeDown(grab, panel, onClose){
  if (!grab || !panel) return;
  let startY = null, dy = 0;
  grab.addEventListener('pointerdown', (e) => {
    if (e.pointerType === 'mouse' || e.target.closest('button, a, input')) return;
    startY = e.clientY; dy = 0;
    grab.setPointerCapture(e.pointerId);
    panel.style.transition = 'none';
  });
  grab.addEventListener('pointermove', (e) => {
    if (startY === null) return;
    dy = Math.max(0, e.clientY - startY);
    panel.style.transform = `translateY(${dy}px)`;
  });
  const end = () => {
    if (startY === null) return;
    startY = null;
    panel.style.transition = '';
    panel.style.transform = '';
    if (dy > 90) onClose();
  };
  grab.addEventListener('pointerup', end);
  grab.addEventListener('pointercancel', end);
}

// ---------------------------------------------------------------- navigare: bară de progres
//
// Paginile se randează pe server; între clic și pagina nouă se vede o bară subțire sus,
// ca să fie clar că s-a apăsat. Nu pornește pentru descărcări (/resurse/…), linkuri externe
// sau cu target, ori când pagina a oprit singură navigarea (preventDefault).
const navProgress = document.createElement('div');
navProgress.id = 'nav-progress';
document.body.appendChild(navProgress);
function startNavProgress(){
  navProgress.classList.remove('run');
  void navProgress.offsetWidth;
  navProgress.classList.add('run');
  clearTimeout(startNavProgress._t);
  startNavProgress._t = setTimeout(stopNavProgress, 12000);
}
function stopNavProgress(){ navProgress.classList.remove('run'); }
window.addEventListener('pageshow', stopNavProgress);
document.addEventListener('click', (e) => {
  const a = e.target.closest && e.target.closest('a[href]');
  if (!a || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
  if (a.target || a.hasAttribute('download') || a.getAttribute('href').startsWith('#')) return;
  const url = new URL(a.href, location.href);
  if (url.origin !== location.origin || /^\/(resurse|api)\//.test(url.pathname)) return;
  setTimeout(() => { if (!e.defaultPrevented) startNavProgress(); }, 0);
});
document.addEventListener('submit', (e) => {
  setTimeout(() => { if (!e.defaultPrevented) startNavProgress(); }, 0);
});

// ---------------------------------------------------------------- tragere în jos = reîncarcă
//
// Doar în aplicația instalată: acolo nu există butonul de reîncărcare al browserului.
function isStandalone(){
  return (window.matchMedia && window.matchMedia('(display-mode: standalone)').matches) || navigator.standalone === true;
}
(function initPullToRefresh(){
  if (!isStandalone()) return;
  const ind = document.createElement('div');
  ind.id = 'ptr';
  ind.innerHTML = '<svg viewBox="0 0 26 26" fill="none"><path d="M1 13H7L10 4L16 22L19 13H25"/></svg>';
  document.body.appendChild(ind);
  let startY = null, armed = false;
  const reset = () => { startY = null; armed = false; ind.style.transform = ''; ind.style.opacity = ''; ind.classList.remove('armed'); };
  window.addEventListener('touchstart', (e) => {
    const busy = document.documentElement.classList.contains('sheet-open') || document.querySelector('.drawer:not([hidden])');
    if (window.scrollY > 0 || e.touches.length !== 1 || busy || e.target.closest('input, textarea, select, dialog')) { startY = null; return; }
    startY = e.touches[0].clientY;
  }, { passive: true });
  window.addEventListener('touchmove', (e) => {
    if (startY === null) return;
    const pull = Math.min((e.touches[0].clientY - startY) * 0.5, 96);
    if (pull <= 0 || window.scrollY > 0) { reset(); return; }
    ind.style.transform = `translate(-50%, ${pull}px)`;
    ind.style.opacity = String(Math.min(1, pull / 56));
    armed = pull >= 64;
    ind.classList.toggle('armed', armed);
  }, { passive: true });
  window.addEventListener('touchend', () => {
    if (startY === null) return;
    if (armed) { ind.classList.add('loading'); startNavProgress(); location.reload(); }
    else reset();
  });
})();

// fetch JSON către API-ul site-ului; aruncă Error cu mesajul serverului la eșec.
async function api(method, url, body, opts){
  if (method !== 'GET' && !navigator.onLine) throw new Error(OFFLINE_MSG);
  const init = { method, headers: {}, credentials: 'same-origin' };
  if (body instanceof Blob) {
    init.body = body;
    init.headers['content-type'] = body.type || 'application/octet-stream';
  } else if (body !== undefined) {
    init.body = JSON.stringify(body);
    init.headers['content-type'] = 'application/json';
  }
  let res;
  try { res = await fetch(url, init); }
  catch (e) { throw new Error(navigator.onLine ? NETWORK_MSG : OFFLINE_MSG); }
  let data = null;
  try { data = await res.json(); } catch(e){}
  if (res.status === 401) { location.href = '/login?next=' + encodeURIComponent(location.pathname); throw new Error('Sesiune expirată.'); }
  if (!res.ok) throw new Error((data && data.error) || `Eroare ${res.status}`);
  return data;
}

// Butonul arată că lucrează (și nu poate fi apăsat de două ori) cât timp rulează fn.
async function withBusy(btn, fn){
  if (btn) { btn.disabled = true; btn.classList.add('busy'); btn.setAttribute('aria-busy', 'true'); }
  try { return await fn(); }
  finally { if (btn) { btn.disabled = false; btn.classList.remove('busy'); btn.removeAttribute('aria-busy'); } }
}

// Mesaj scurt jos pe ecran (salvat / eroare). Ca popover, apare și deasupra unui panou deschis.
function toast(message, isError){
  let el = document.getElementById('toast');
  if (!el) {
    el = document.createElement('div');
    el.id = 'toast';
    el.setAttribute('role', 'status');
    el.setAttribute('aria-live', 'polite');
    if (el.showPopover) el.setAttribute('popover', 'manual');
    document.body.appendChild(el);
  }
  el.textContent = message;
  if (el.showPopover) { try { el.hidePopover(); } catch(e){} el.showPopover(); }
  el.className = isError ? 'error' : '';
  requestAnimationFrame(() => el.classList.add('show'));
  clearTimeout(toast._t);
  toast._t = setTimeout(() => {
    el.classList.remove('show');
    setTimeout(() => { if (!el.classList.contains('show') && el.hidePopover) { try { el.hidePopover(); } catch(e){} } }, 200);
  }, isError ? 5000 : 2200);
}

// ---------------------------------------------------------------- dialoguri

// În locul lui confirm()/prompt()/alert(): arată ca restul aplicației (și în aplicația
// instalată), iar butonul sigur („Renunță") primește focusul. Toate întorc un Promise.
function confirmDialog(message, opts){
  opts = opts || {};
  return new Promise((resolve) => {
    const dlg = openSheet(`
      <p class="dialog-text">${esc(message)}</p>
      <div class="dialog-actions">
        <button class="btn" type="button" data-v="0">${esc(opts.cancel || 'Renunță')}</button>
        <button class="btn ${opts.danger ? 'danger-solid' : 'primary'}" type="button" data-v="1">${esc(opts.ok || 'Confirmă')}</button>
      </div>`, { title: opts.title || 'Ești sigur?', className: 'dialog', onClose: (v) => resolve(v === true) });
    dlg.querySelectorAll('[data-v]').forEach(b => b.addEventListener('click', () => closeSheet(dlg, b.dataset.v === '1')));
  });
}
function promptDialog(message, opts){
  opts = opts || {};
  return new Promise((resolve) => {
    const dlg = openSheet(`
      <form class="dialog-form" method="dialog">
        ${message ? `<p class="dialog-text">${esc(message)}</p>` : ''}
        <label class="field"><span>${esc(opts.label || 'Valoare')}</span>
          <input class="input" name="v" value="${esc(opts.value || '')}" autocomplete="off" autocapitalize="none" spellcheck="false"${opts.minLength ? ` minlength="${opts.minLength}"` : ''} required></label>
        <div class="dialog-actions">
          <button class="btn" type="button" data-cancel>Renunță</button>
          <button class="btn primary" type="submit">${esc(opts.ok || 'Salvează')}</button>
        </div>
      </form>`, { title: opts.title, className: 'dialog', onClose: (v) => resolve(typeof v === 'string' ? v : null) });
    const form = dlg.querySelector('form');
    dlg.querySelector('[data-cancel]').addEventListener('click', () => closeSheet(dlg, null));
    form.addEventListener('submit', (e) => { e.preventDefault(); closeSheet(dlg, form.v.value); });
    const input = form.v;
    setTimeout(() => { input.focus(); input.select(); }, 50);
  });
}
// Date de dat mai departe (utilizator + parolă): fiecare pe rândul ei, cu „Copiază tot".
function credentialsDialog(title, rows, note){
  return new Promise((resolve) => {
    const text = rows.map(r => `${r.label}: ${r.value}`).join('\n');
    const dlg = openSheet(`
      <div class="cred-list">${rows.map(r => `<div class="cred-row"><span>${esc(r.label)}</span><code>${esc(r.value)}</code></div>`).join('')}</div>
      ${note ? `<p class="dialog-note">${esc(note)}</p>` : ''}
      <div class="dialog-actions">
        <button class="btn" type="button" data-copy>Copiază tot</button>
        <button class="btn primary" type="button" data-ok>Gata</button>
      </div>`, { title, className: 'dialog', onClose: () => resolve() });
    dlg.querySelector('[data-ok]').addEventListener('click', () => closeSheet(dlg));
    dlg.querySelector('[data-copy]').addEventListener('click', async (e) => {
      const btn = e.currentTarget;
      try { await navigator.clipboard.writeText(text); btn.textContent = 'Copiat ✓'; }
      catch (err) { // fără acces la clipboard: selectăm textul, se poate copia manual
        const sel = window.getSelection(); const range = document.createRange();
        range.selectNodeContents(dlg.querySelector('.cred-list')); sel.removeAllRanges(); sel.addRange(range);
        btn.textContent = 'Selectat — copiază manual';
      }
    });
  });
}

// Bandă sus cât timp nu e conexiune; modificările sunt oprite de api() cu un mesaj clar.
// PULS_SAVED_AT e pus de service worker (public/sw.js) când pagina vine din copia salvată
// pe dispozitiv, pentru că serverul n-a putut fi contactat.
function savedAtLabel(){
  if (typeof window.PULS_SAVED_AT !== 'string' || !window.PULS_SAVED_AT) return '';
  const d = new Date(window.PULS_SAVED_AT);
  if (isNaN(d)) return '';
  const pad = (n) => String(n).padStart(2, '0');
  const today = new Date().toDateString() === d.toDateString();
  return (today ? 'azi' : `${d.getDate()} ${MONTHS_RO[d.getMonth()]}`) + ` la ${pad(d.getHours())}:${pad(d.getMinutes())}`;
}
function updateOnlineState(){
  const saved = savedAtLabel();
  const offline = !navigator.onLine || !!saved;
  document.documentElement.classList.toggle('is-offline', offline);
  let banner = document.getElementById('offline-banner');
  if (offline && !banner) {
    banner = document.createElement('div');
    banner.id = 'offline-banner';
    banner.setAttribute('role', 'status');
    const top = document.getElementById('topbar-root');
    if (top && top.parentNode) top.after(banner); else document.body.prepend(banner);
  }
  if (banner) {
    banner.textContent = !saved
      ? 'Ești offline — vezi ultima versiune încărcată. Modificările nu se pot salva până revine conexiunea.'
      : window.PULS_SAVED_REASON === 'slow' && navigator.onLine
        ? `Conexiune slabă — versiunea salvată ${saved}. Trage în jos sau reîncarcă pentru cea nouă.`
        : `Offline — versiunea salvată ${saved}. Modificările nu se pot salva până revine conexiunea.`;
    banner.hidden = !offline;
  }
}
window.addEventListener('online', updateOnlineState);
window.addEventListener('offline', updateOnlineState);
setTimeout(updateOnlineState, 0);

// Lățimea reală a unui SVG, ca graficele să fie desenate la mărimea ecranului (textul
// rămâne lizibil pe telefon, în loc să fie micșorat cu tot desenul).
function chartWidth(svg, fallback){
  const w = Math.round(svg.getBoundingClientRect().width);
  return w > 0 ? Math.max(300, w) : fallback;
}
// Rulează fn când se schimbă lățimea elementului (rotire telefon, fereastră redimensionată).
function onWidthChange(el, fn){
  if (!('ResizeObserver' in window)) return;
  let last = el.getBoundingClientRect().width, t = null;
  new ResizeObserver(([entry]) => {
    const w = entry.contentRect.width;
    if (Math.abs(w - last) < 8) return;
    last = w;
    clearTimeout(t); t = setTimeout(fn, 120);
  }).observe(el);
}

function remember(key, value){ try { localStorage.setItem('puls-' + key, value); } catch(e){} }
function recall(key){ try { return localStorage.getItem('puls-' + key); } catch(e){ return null; } }

// "2026-09-27" -> "duminică, 27 septembrie 2026"
function formatSlugLong(slug){
  const [y,m,d] = slug.split('-').map(Number);
  return `${d} ${MONTHS_RO_FULL[m-1]} ${y}`;
}
function formatDuration(sec){
  sec = Math.max(0, Math.round(sec||0));
  const h = Math.floor(sec/3600), m = Math.floor((sec%3600)/60), s = sec%60;
  if (h) return `${h}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`;
  return `${m}:${String(s).padStart(2,'0')}`;
}

let installPrompt = null;
function isIOS(){
  return /iphone|ipad|ipod/i.test(navigator.userAgent) || (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1);
}
function canOfferInstall(){ return !isStandalone() && (installPrompt || isIOS()); }

async function runInstall(){
  if (installPrompt) {
    const p = installPrompt;
    installPrompt = null;
    p.prompt();
    try { await p.userChoice; } catch(e){}
    hideInstallHint();
    return;
  }
  if (isIOS()) showIosInstallHelp();
}
function showIosInstallHelp(){
  const share = '<svg class="ios-share" viewBox="0 0 24 24" aria-label="Share"><path d="M12 3v12M8 7l4-4 4 4M5 11v8.5A1.5 1.5 0 0 0 6.5 21h11a1.5 1.5 0 0 0 1.5-1.5V11"/></svg>';
  const dlg = openSheet(`
    <ol class="ios-steps">
      <li>Apasă butonul <b>Share</b> ${share} din bara Safari.</li>
      <li>Derulează și alege <b>Adaugă pe ecranul principal</b>.</li>
      <li>Apasă <b>Adaugă</b> — Pulsul apare lângă celelalte aplicații și se deschide pe tot ecranul.</li>
    </ol>
    <div class="dialog-actions"><button class="btn primary" type="button" data-ok>Am înțeles</button></div>`,
    { title: 'Instalează pe iPhone', className: 'dialog' });
  dlg.querySelector('[data-ok]').addEventListener('click', () => { remember('install-hint', 'no'); hideInstallHint(); closeSheet(dlg); });
}

// Rândul din panoul de cont (apelat de openAccountSheet).
function fillInstallSlot(slot){
  if (!slot || !canOfferInstall()) return;
  slot.innerHTML = `<button class="sheet-row" type="button">${icon('install')}<span class="sr-label">${installPrompt ? 'Instalează aplicația' : 'Instalează pe iPhone'}</span><span class="chev" aria-hidden="true">›</span></button>`;
  slot.firstElementChild.addEventListener('click', () => {
    const dlg = slot.closest('dialog');
    if (dlg) closeSheet(dlg);
    setTimeout(runInstall, 200);
  });
}

// Sugestia de instalare: o singură dată (până e închisă), doar pentru cine e logat.
function showInstallHint(){
  if (!canOfferInstall() || recall('install-hint') === 'no' || typeof PULS_USER === 'undefined' || !PULS_USER) return;
  if (document.getElementById('install-hint')) return;
  const el = document.createElement('div');
  el.id = 'install-hint';
  el.setAttribute('role', 'dialog');
  el.setAttribute('aria-label', 'Instalează aplicația');
  el.innerHTML = `
    <img src="/icons/icon-192.png" alt="" width="40" height="40">
    <div class="ih-text"><b>Pulsul, ca aplicație</b><span>Pe ecranul principal, pe tot ecranul, și fără semnal.</span></div>
    <button class="btn primary small" type="button" data-install>${installPrompt ? 'Instalează' : 'Cum?'}</button>
    <button class="ih-close" type="button" aria-label="Nu acum">×</button>`;
  document.body.appendChild(el);
  el.querySelector('[data-install]').addEventListener('click', runInstall);
  el.querySelector('.ih-close').addEventListener('click', () => { remember('install-hint', 'no'); hideInstallHint(); });
}
function hideInstallHint(){ const el = document.getElementById('install-hint'); if (el) el.remove(); }

window.addEventListener('beforeinstallprompt', (e) => {
  e.preventDefault();
  installPrompt = e;
  document.querySelectorAll('.menu-install').forEach(b => { b.hidden = false; });
  showInstallHint();
});
window.addEventListener('appinstalled', () => { installPrompt = null; hideInstallHint(); remember('install-hint', 'no'); });
if (isIOS()) setTimeout(showInstallHint, 1500);
